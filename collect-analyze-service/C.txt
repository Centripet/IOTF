#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <math.h>
#include <stdbool.h>
#define MAX_DEVICE_NUM 32          // 最大设备数量
#define DEFAULT_SAMPLING_INTERVAL 60  // 默认采样间隔(秒)
#define MIN_SAMPLING_INTERVAL 10    // 最小采样间隔(秒)
#define MAX_SAMPLING_INTERVAL 300   // 最大采样间隔(秒)
#define MAX_DATA_BUFFER 1024        // 数据缓冲区大小
#define MAX_RETRANSMIT_COUNT 3      // 最大重传次数
#define ENERGY_THRESHOLD_HIGH 500   // 高能耗报警阈值(Wh)
#define CURRENT_THRESHOLD 10        // 电流过载阈值(A)
#define VOLTAGE_NORMAL_MIN 200      // 正常电压范围最小值(V)
#define VOLTAGE_NORMAL_MAX 240      // 正常电压范围最大值(V)
#define BACKUP_INTERVAL 86400       // 数据备份间隔(秒，24小时)
typedef enum {
    DEVICE_LIGHT,       // 照明设备
    DEVICE_AIRCON,      // 空调
    DEVICE_REFRIGERATOR,// 冰箱
    DEVICE_TV,          // 电视
    DEVICE_WATER_HEATER,// 热水器
    DEVICE_OTHER        // 其他设备
} DeviceType;
typedef enum {
    COMM_WIFI,          // WiFi通讯
    COMM_BLUETOOTH,     // 蓝牙通讯
    COMM_ZIGBEE         // Zigbee通讯
} CommType;
typedef enum {
    ALARM_OVERLOAD,     // 过载报警
    ALARM_HIGH_ENERGY,  // 高能耗报警
    ALARM_ELECTRIC_LEAK,// 漏电报警
    ALARM_DEVICE_FAULT, // 设备故障报警
    ALARM_NONE          // 无报警
} AlarmType;
typedef struct {
    bool isOn;          // 开关状态
    float current;      // 电流(A)
    float voltage;      // 电压(V)
    float power;        // 功率(W)
    float energy;       // 用电量(Wh)
    time_t lastUpdate;  // 最后更新时间
    bool isFault;       // 故障状态
} DeviceStatus;
typedef struct {
    char id[32];        // 设备ID
    char name[64];      // 设备名称
    DeviceType type;    // 设备类型
    char location[64];  // 安装位置
    DeviceStatus status;// 设备状态
} DeviceInfo;
typedef struct {
    char deviceId[32];  // 设备ID
    time_t timestamp;   // 时间戳
    float current;      // 电流(A)
    float voltage;      // 电压(V)
    float power;        // 功率(W)
    float energy;       // 用电量(Wh)
    bool isComplete;    // 数据完整性标志
} EnergyData;
typedef struct {
    AlarmType type;     // 报警类型
    char deviceId[32];  // 关联设备ID
    time_t timestamp;   // 报警时间
    char description[128];// 报警描述
    bool isHandled;     // 处理状态
} AlarmInfo;
typedef struct {
    int samplingInterval;   // 采样间隔(秒)
    float overloadThreshold;// 过载阈值(W)
    float highEnergyThreshold;// 高能耗阈值(Wh)
    bool notifyByApp;       // APP通知开关
    bool notifyBySms;       // 短信通知开关
    char mqttServer[128];   // MQTT服务器地址
} UserConfig;
typedef struct {
    DeviceInfo devices[MAX_DEVICE_NUM];  // 设备列表
    int deviceCount;                     // 设备数量
    EnergyData dataBuffer[MAX_DATA_BUFFER];// 数据缓冲区
    int dataBufferIndex;                 // 缓冲区索引
    AlarmInfo alarms[128];               // 报警列表
    int alarmCount;                      // 报警数量
    UserConfig config;                   // 用户配置
    time_t lastBackupTime;               // 最后备份时间
    bool isRunning;                      // 系统运行状态
} SystemState;
SystemState sysState;
void initSystem() {
    memset(&sysState, 0, sizeof(SystemState));
    sysState.deviceCount = 0;
    sysState.dataBufferIndex = 0;
    sysState.alarmCount = 0;
    sysState.lastBackupTime = time(NULL);
    sysState.isRunning = true;
    sysState.config.samplingInterval = DEFAULT_SAMPLING_INTERVAL;
    sysState.config.overloadThreshold = 2000;  // 默认过载阈值2000W
    sysState.config.highEnergyThreshold = ENERGY_THRESHOLD_HIGH;
    sysState.config.notifyByApp = true;
    sysState.config.notifyBySms = true;
    strcpy(sysState.config.mqttServer, "mqtt://energy-server.local:1883");
    printf("系统初始化完成\n");
}
bool addDevice(const char* id, const char* name, DeviceType type, const char* location) {
    if (sysState.deviceCount >= MAX_DEVICE_NUM) {
        printf("设备数量已达上限\n");
        return false;
    }

    DeviceInfo* dev = &sysState.devices[sysState.deviceCount];
    strcpy(dev->id, id);
    strcpy(dev->name, name);
    dev->type = type;
    strcpy(dev->location, location);
    dev->status.isOn = false;
    dev->status.current = 0;
    dev->status.voltage = 0;
    dev->status.power = 0;
    dev->status.energy = 0;
    dev->status.lastUpdate = time(NULL);
    dev->status.isFault = false;

    sysState.deviceCount++;
    printf("设备添加成功: %s(%s)\n", name, id);
    return true;
}

// 查找设备索引
int findDeviceIndex(const char* deviceId) {
    for (int i = 0; i < sysState.deviceCount; i++) {
        if (strcmp(sysState.devices[i].id, deviceId) == 0) {
            return i;
        }
    }
    return -1;
}

// 传感器自检
bool sensorSelfCheck(const char* deviceId) {
    int index = findDeviceIndex(deviceId);
    if (index == -1) return false;

    // 传感器自检: 95%概率正常
    float randVal = (float)rand() / RAND_MAX;
    bool result = (randVal < 0.95);
    sysState.devices[index].status.isFault = !result;
    
    if (!result) {
        printf("设备%s传感器自检\n", deviceId);
    }
    return result;
}

// 采集单设备数据(传感器数据)
DeviceStatus collectDeviceData(const char* deviceId) {
    int index = findDeviceIndex(deviceId);
    DeviceStatus status = {0};
    if (index == -1) return status;

    DeviceInfo* dev = &sysState.devices[index];
    status.isOn = (rand() % 10 < 8);  // 80%概率开机状态

    if (status.isOn) { 
        switch (dev->type) {
            case DEVICE_LIGHT:
                status.power = 10 + (float)(rand() % 50);  // 10-60W
                break;
            case DEVICE_AIRCON:
                status.power = 800 + (float)(rand() % 1200); // 800-2000W
                break;
            case DEVICE_REFRIGERATOR:
                status.power = 100 + (float)(rand() % 150); // 100-250W
                break;
            case DEVICE_TV:
                status.power = 50 + (float)(rand() % 150);  // 50-200W
                break;
            case DEVICE_WATER_HEATER:
                status.power = 1500 + (float)(rand() % 1500); // 1500-3000W
                break;
            default:
                status.power = 50 + (float)(rand() % 250);  // 50-300W
        }

        // 生成电压(220V左右波动)
        status.voltage = 220 + (float)(rand() % 21) - 10;
        // 电流 = 功率 / 电压
        status.current = status.power / status.voltage;
        // 用电量(基于采样间隔计算)
        status.energy = status.power * sysState.config.samplingInterval / 3600;
    } else {
        status.power = 0;
        status.current = 0;
        status.voltage = 0;
        status.energy = 0;
    }

    status.lastUpdate = time(NULL);
    status.isFault = false;

    // 1%概率漏电
    if (rand() % 100 < 1) {
        status.current += 5;  // 漏电导致电流异常升高
    }

    return status;
}

// 数据滤波处理(简单滑动平均)
float filterData(float newVal, float lastVal) {
    return 0.7 * newVal + 0.3 * lastVal;  // 新值权重70%，旧值30%
}

// 数据格式转换
void convertDataFormat(EnergyData* data) {
    // 保留两位小数
    data->current = roundf(data->current * 100) / 100;
    data->voltage = roundf(data->voltage * 100) / 100;
    data->power = roundf(data->power * 100) / 100;
    data->energy = roundf(data->energy * 100) / 100;
}

// 数据采集处理
void collectEnergyData() {
    printf("\n开始数据采集...\n");
    for (int i = 0; i < sysState.deviceCount; i++) {
        DeviceInfo* dev = &sysState.devices[i];
        printf("处理设备: %s ", dev->name);

        // 传感器自检
        if (!sensorSelfCheck(dev->id)) {
            // 传感器故障，记录故障信息
            dev->status.isFault = true;
            printf("[传感器故障]\n");
            continue;
        }

        // 采集设备数据
        DeviceStatus newStatus = collectDeviceData(dev->id);
        
        // 数据滤波处理
        newStatus.current = filterData(newStatus.current, dev->status.current);
        newStatus.voltage = filterData(newStatus.voltage, dev->status.voltage);
        newStatus.power = filterData(newStatus.power, dev->status.power);

        // 更新设备状态
        dev->status = newStatus;

        // 生成能耗数据记录
        EnergyData data;
        strcpy(data.deviceId, dev->id);
        data.timestamp = time(NULL);
        data.current = newStatus.current;
        data.voltage = newStatus.voltage;
        data.power = newStatus.power;
        data.energy = newStatus.energy;
        data.isComplete = true;

        // 数据格式转换
        convertDataFormat(&data);

        // 存入数据缓冲区
        if (sysState.dataBufferIndex < MAX_DATA_BUFFER) {
            sysState.dataBuffer[sysState.dataBufferIndex] = data;
            sysState.dataBufferIndex++;
        } else {
            // 缓冲区满时覆盖最早数据
            memmove(sysState.dataBuffer, sysState.dataBuffer + 1, 
                   (MAX_DATA_BUFFER - 1) * sizeof(EnergyData));
            sysState.dataBuffer[MAX_DATA_BUFFER - 1] = data;
        }

        printf("[采集完成, 功率: %.2fW, 用电量: %.2fWh]\n", 
               data.power, data.energy);
    }
    printf("数据采集结束\n");
}

// 检查通讯方式可用性
bool isCommAvailable(CommType type) {
    // 通讯方式可用性: WiFi和Zigbee可靠性高，蓝牙稍低
    switch (type) {
        case COMM_WIFI:
            return (rand() % 100 < 98);  // 98%可用
        case COMM_BLUETOOTH:
            return (rand() % 100 < 90);  // 90%可用
        case COMM_ZIGBEE:
            return (rand() % 100 < 95);  // 95%可用
        default:
            return false;
    }
}

// 选择合适的通讯方式
CommType selectCommType(const char* deviceId) {
    // 根据设备类型选择通讯方式
    int index = findDeviceIndex(deviceId);
    if (index == -1) return COMM_WIFI;  // 默认WiFi

    // 高功耗设备优先用WiFi，低功耗用Zigbee
    if (sysState.devices[index].type == DEVICE_AIRCON || 
        sysState.devices[index].type == DEVICE_WATER_HEATER) {
        if (isCommAvailable(COMM_WIFI)) return COMM_WIFI;
    }
    
    // 优先检查Zigbee
    if (isCommAvailable(COMM_ZIGBEE)) return COMM_ZIGBEE;
    
    // 最后检查蓝牙
    if (isCommAvailable(COMM_BLUETOOTH)) return COMM_BLUETOOTH;
    
    // 都不可用时返回WiFi(作为默认)
    return COMM_WIFI;
}

// 数据加密处理(简单XOR加密)
void encryptData(char* data, int len, const char* key) {
    int keyLen = strlen(key);
    for (int i = 0; i < len; i++) {
        data[i] ^= key[i % keyLen];
    }
}

// 数据传输
bool transmitData(EnergyData* data) {
    if (!data->isComplete) {
        printf("数据不完整，无法传输\n");
        return false;
    }

    // 选择通讯方式
    CommType commType = selectCommType(data->deviceId);
    const char* commName = NULL;
    switch (commType) {
        case COMM_WIFI: commName = "WiFi"; break;
        case COMM_BLUETOOTH: commName = "蓝牙"; break;
        case COMM_ZIGBEE: commName = "Zigbee"; break;
    }

    // 检查通讯方式是否可用
    if (!isCommAvailable(commType)) {
        printf("通讯方式%s不可用，尝试切换...\n", commName);
        // 尝试切换其他通讯方式
        for (int i = 0; i < 3; i++) {
            CommType newType = (commType + i + 1) % 3;
            if (isCommAvailable(newType)) {
                commType = newType;
                switch (commType) {
                    case COMM_WIFI: commName = "WiFi"; break;
                    case COMM_BLUETOOTH: commName = "蓝牙"; break;
                    case COMM_ZIGBEE: commName = "Zigbee"; break;
                }
                printf("切换至%s通讯\n", commName);
                break;
            }
        }
        if (!isCommAvailable(commType)) {
            printf("所有通讯方式均不可用\n");
            return false;
        }
    }

    // 格式化数据
    char dataStr[256];
    sprintf(dataStr, "ID:%s,TIME:%ld,CURR:%.2f,VOLT:%.2f,POWER:%.2f,ENERGY:%.2f",
            data->deviceId, data->timestamp,
            data->current, data->voltage,
            data->power, data->energy);

    // 数据加密
    encryptData(dataStr, strlen(dataStr), "energy_key");

    // 数据传输
    printf("通过%s传输数据: %s\n", commName, dataStr);

    // 传输结果(98%成功率)
    bool success = (rand() % 100 < 98);
    if (!success) {
        // 重传机制
        for (int i = 0; i < MAX_RETRANSMIT_COUNT; i++) {
            printf("传输，第%d次重传...\n", i + 1);
            success = (rand() % 100 < 98);
            if (success) {
                printf("重传成功\n");
                break;
            }
        }
        if (!success) {
            printf("达到最大重传次数，传输\n");
        }
    } else {
        printf("数据传输成功\n");
    }

    return success;
}

// 传输所有缓冲数据
void transmitAllData() {
    printf("\n开始数据传输...\n");
    for (int i = 0; i < sysState.dataBufferIndex; i++) {
        printf("传输第%d条数据: ", i + 1);
        transmitData(&sysState.dataBuffer[i]);
    }
    // 传输完成后清空缓冲区
    sysState.dataBufferIndex = 0;
    printf("数据传输结束\n");
}

// 数据清洗: 去除异常值
void cleanData(EnergyData* data, int count) {
    for (int i = 0; i < count; i++) {
        // 电流异常值处理
        if (data[i].current < 0 || data[i].current > 30) {
            printf("发现异常电流值 %.2fA，已修正\n", data[i].current);
            data[i].current = 0;
            data[i].isComplete = false;
        }
        // 电压异常值处理
        if (data[i].voltage < VOLTAGE_NORMAL_MIN || data[i].voltage > VOLTAGE_NORMAL_MAX) {
            printf("发现异常电压值 %.2fV，已修正\n", data[i].voltage);
            data[i].voltage = 220;  // 修正为标准电压
        }
        // 功率异常值处理
        if (data[i].power < 0 || data[i].power > 5000) {
            printf("发现异常功率值 %.2fW，已修正\n", data[i].power);
            data[i].power = 0;
            data[i].isComplete = false;
        }
    }
}

// 填补缺失数据
void fillMissingData(EnergyData* data, int count) {
    for (int i = 0; i < count; i++) {
        if (!data[i].isComplete && i > 0) {
            // 用前一条数据填补
            data[i].current = data[i-1].current;
            data[i].voltage = data[i-1].voltage;
            data[i].power = data[i-1].power * 0.9;  // 略低于前值
            data[i].energy = data[i-1].energy * 0.9;
            data[i].isComplete = true;
            printf("填补缺失数据: 设备%s\n", data[i].deviceId);
        }
    }
}

// 统计设备日能耗
float calculateDailyEnergy(const char* deviceId) {
    time_t now = time(NULL);
    struct tm* tmNow = localtime(&now);
    tmNow->tm_hour = 0;
    tmNow->tm_min = 0;
    tmNow->tm_sec = 0;
    time_t todayStart = mktime(tmNow);

    float total = 0;
    // 从缓冲区统计(实际应从数据库查询)
    for (int i = 0; i < sysState.dataBufferIndex; i++) {
        if (strcmp(sysState.dataBuffer[i].deviceId, deviceId) == 0 &&
            sysState.dataBuffer[i].timestamp >= todayStart) {
            total += sysState.dataBuffer[i].energy;
        }
    }
    return total;
}

// 统计设备周能耗
float calculateWeeklyEnergy(const char* deviceId) {
    time_t now = time(NULL);
    struct tm* tmNow = localtime(&now);
    int dayOfWeek = tmNow->tm_wday;
    time_t weekStart = now - dayOfWeek * 86400;
    tmNow = localtime(&weekStart);
    tmNow->tm_hour = 0;
    tmNow->tm_min = 0;
    tmNow->tm_sec = 0;
    weekStart = mktime(tmNow);

    float total = 0;
    // 从缓冲区统计(实际应从数据库查询)
    for (int i = 0; i < sysState.dataBufferIndex; i++) {
        if (strcmp(sysState.dataBuffer[i].deviceId, deviceId) == 0 &&
            sysState.dataBuffer[i].timestamp >= weekStart) {
            total += sysState.dataBuffer[i].energy;
        }
    }
    return total;
}

// 分析设备能耗占比
void analyzeEnergyRatio() {
    printf("\n开始能耗占比分析...\n");
    float totalEnergy = 0;
    float deviceEnergy[MAX_DEVICE_NUM] = {0};

    // 计算总能耗和各设备能耗
    for (int i = 0; i < sysState.dataBufferIndex; i++) {
        EnergyData* data = &sysState.dataBuffer[i];
        int devIndex = findDeviceIndex(data->deviceId);
        if (devIndex != -1) {
            deviceEnergy[devIndex] += data->energy;
            totalEnergy += data->energy;
        }
    }

    // 输出各设备能耗占比
    if (totalEnergy > 0) {
        for (int i = 0; i < sysState.deviceCount; i++) {
            DeviceInfo* dev = &sysState.devices[i];
            float ratio = (deviceEnergy[i] / totalEnergy) * 100;
            printf("%s(%s): 能耗 %.2fWh, 占比 %.2f%%\n",
                   dev->name, dev->id, deviceEnergy[i], ratio);
        }
    } else {
        printf("无有效能耗数据可分析\n");
    }
    printf("能耗占比分析结束\n");
}

// 分析能耗趋势
void analyzeEnergyTrend(const char* deviceId) {
    printf("\n开始能耗趋势分析...\n");
    int index = findDeviceIndex(deviceId);
    if (index == -1) {
        printf("设备不存在\n");
        return;
    }

    DeviceInfo* dev = &sysState.devices[index];
    float daily = calculateDailyEnergy(deviceId);
    float weekly = calculateWeeklyEnergy(deviceId);
    
    // 简单趋势预测: 基于当前数据推算
    float monthlyPred = daily * 30;
    float yearlyPred = weekly * 52;

    printf("%s趋势分析:\n", dev->name);
    printf("今日能耗: %.2fWh\n", daily);
    printf("本周能耗: %.2fWh\n", weekly);
    printf("预计月能耗: %.2fWh\n", monthlyPred);
    printf("预计年能耗: %.2fWh\n", yearlyPred);

    // 识别高能耗时段()
    printf("高能耗时段: 14:00-16:00, 19:00-21:00\n");
    
    // 生成节能建议
    printf("节能建议: \n");
    switch (dev->type) {
        case DEVICE_AIRCON:
            printf("- 温度调高1-2℃可降低10-15%%能耗\n");
            printf("- 避免频繁开关空调\n");
            break;
        case DEVICE_WATER_HEATER:
            printf("- 设置合理温度(50-55℃)，避免过热\n");
            printf("- 使用保温模式减少反复加热\n");
            break;
        case DEVICE_REFRIGERATOR:
            printf("- 保持冰箱内物品充足，减少开门次数\n");
            printf("- 定期清理冷凝器灰尘\n");
            break;
        default:
            printf("- 无人时及时关闭设备电源\n");
            printf("- 选择节能模式运行\n");
    }
    printf("能耗趋势分析结束\n");
}

// 生成能耗报表
void generateEnergyReport() {
    printf("\n===== 家庭能耗报表 =====\n");
    printf("报表生成时间: %s", ctime(&sysState.dataBuffer[0].timestamp));
    
    // 计算总能耗
    float totalEnergy = 0;
    for (int i = 0; i < sysState.dataBufferIndex; i++) {
        totalEnergy += sysState.dataBuffer[i].energy;
    }
    printf("总能耗: %.2fWh\n", totalEnergy);
    
    // 各设备日能耗
    printf("\n设备日能耗统计:\n");
    for (int i = 0; i < sysState.deviceCount; i++) {
        DeviceInfo* dev = &sysState.devices[i];
        float daily = calculateDailyEnergy(dev->id);
        printf("- %s: %.2fWh\n", dev->name, daily);
    }
    
    // 高能耗设备识别
    float maxEnergy = 0;
    int maxIndex = -1;
    for (int i = 0; i < sysState.deviceCount; i++) {
        float energy = calculateDailyEnergy(sysState.devices[i].id);
        if (energy > maxEnergy) {
            maxEnergy = energy;
            maxIndex = i;
        }
    }
    if (maxIndex != -1) {
        printf("\n高能耗设备: %s (%.2fWh)\n", 
               sysState.devices[maxIndex].name, maxEnergy);
    }
}

// 数据分析处理
void analyzeEnergyData() {
    printf("\n开始数据分析...\n");
    
    // 数据清洗
    cleanData(sysState.dataBuffer, sysState.dataBufferIndex);
    
    // 填补缺失数据
    fillMissingData(sysState.dataBuffer, sysState.dataBufferIndex);
    
    // 能耗占比分析
    analyzeEnergyRatio();
    
    // 对每个设备进行趋势分析
    for (int i = 0; i < sysState.deviceCount; i++) {
        analyzeEnergyTrend(sysState.devices[i].id);
    }
    
    // 生成能耗报表
    generateEnergyReport();
    
    printf("数据分析结束\n");
}

// 添加报警信息
void addAlarm(AlarmType type, const char* deviceId, const char* description) {
    if (sysState.alarmCount >= 128) {
        // 报警列表满时移除最早的报警
        memmove(sysState.alarms, sysState.alarms + 1, 127 * sizeof(AlarmInfo));
        sysState.alarmCount--;
    }

    AlarmInfo* alarm = &sysState.alarms[sysState.alarmCount];
    alarm->type = type;
    strcpy(alarm->deviceId, deviceId);
    alarm->timestamp = time(NULL);
    strcpy(alarm->description, description);
    alarm->isHandled = false;

    sysState.alarmCount++;
    printf("新增报警: %s\n", description);
}

// 通知用户(多渠道)
void notifyUser(AlarmInfo* alarm) {
    const char* alarmTypeStr = NULL;
    switch (alarm->type) {
        case ALARM_OVERLOAD: alarmTypeStr = "设备过载"; break;
        case ALARM_HIGH_ENERGY: alarmTypeStr = "异常高能耗"; break;
        case ALARM_ELECTRIC_LEAK: alarmTypeStr = "漏电警告"; break;
        case ALARM_DEVICE_FAULT: alarmTypeStr = "设备故障"; break;
        default: alarmTypeStr = "未知报警";
    }

    // APP通知
    if (sysState.config.notifyByApp) {
        printf("[APP通知] %s: %s (%s)\n", 
               alarmTypeStr, alarm->description, alarm->deviceId);
    }

    // 短信通知
    if (sysState.config.notifyBySms) {
        printf("[短信通知] 家庭能耗系统报警: %s - %s\n", 
               alarmTypeStr, alarm->description);
    }
}

// 检查设备过载
void checkOverload() {
    for (int i = 0; i < sysState.deviceCount; i++) {
        DeviceInfo* dev = &sysState.devices[i];
        if (dev->status.isOn && dev->status.power > sysState.config.overloadThreshold) {
            char desc[128];
            sprintf(desc, "%s功率超过阈值(%.2fW > %.2fW)", 
                    dev->name, dev->status.power, sysState.config.overloadThreshold);
            addAlarm(ALARM_OVERLOAD, dev->id, desc);
        }
    }
}

// 检查异常高能耗
void checkHighEnergy() {
    time_t now = time(NULL);
    struct tm* tmNow = localtime(&now);
    tmNow->tm_hour = 0;
    tmNow->tm_min = 0;
    tmNow->tm_sec = 0;
    time_t todayStart = mktime(tmNow);

    for (int i = 0; i < sysState.deviceCount; i++) {
        DeviceInfo* dev = &sysState.devices[i];
        float dailyEnergy = calculateDailyEnergy(dev->id);
        if (dailyEnergy > sysState.config.highEnergyThreshold) {
            char desc[128];
            sprintf(desc, "%s日能耗过高(%.2fWh > %.2fWh)", 
                    dev->name, dailyEnergy, sysState.config.highEnergyThreshold);
            addAlarm(ALARM_HIGH_ENERGY, dev->id, desc);
        }
    }
}

// 检查漏电情况
void checkElectricLeak() {
    for (int i = 0; i < sysState.deviceCount; i++) {
        DeviceInfo* dev = &sysState.devices[i];
        // 漏电判断: 电流异常且功率与电流不匹配
        if (dev->status.current > CURRENT_THRESHOLD && 
            dev->status.power < dev->status.current * dev->status.voltage * 0.5) {
            char desc[128];
            sprintf(desc, "%s可能存在漏电(电流异常: %.2fA)", 
                    dev->name, dev->status.current);
            addAlarm(ALARM_ELECTRIC_LEAK, dev->id, desc);
        }
    }
}

// 检查设备故障
void checkDeviceFault() {
    for (int i = 0; i < sysState.deviceCount; i++) {
        DeviceInfo* dev = &sysState.devices[i];
        if (dev->status.isFault) {
            char desc[128];
            sprintf(desc, "%s传感器故障，请检查设备", dev->name);
            addAlarm(ALARM_DEVICE_FAULT, dev->id, desc);
        }
    }
}

// 处理报警
void processAlarms() {
    printf("\n开始报警检测...\n");
    
    // 各类报警检查
    checkOverload();
    checkHighEnergy();
    checkElectricLeak();
    checkDeviceFault();
    
    // 通知用户未处理的报警
    for (int i = 0; i < sysState.alarmCount; i++) {
        AlarmInfo* alarm = &sysState.alarms[i];
        if (!alarm->isHandled) {
            notifyUser(alarm);
        }
    }
    
    printf("报警检测结束\n");
}

// 标记报警为已处理
void handleAlarm(int alarmIndex) {
    if (alarmIndex >= 0 && alarmIndex < sysState.alarmCount) {
        sysState.alarms[alarmIndex].isHandled = true;
        printf("报警已处理: %s\n", sysState.alarms[alarmIndex].description);
    }
}

// 保存数据到文件(数据库存储)
bool saveDataToFile(EnergyData* data, const char* filename) {
    FILE* fp = fopen(filename, "a");
    if (!fp) {
        printf("无法打开数据文件\n");
        return false;
    }

    fprintf(fp, "%s,%ld,%.2f,%.2f,%.2f,%.2f,%d\n",
            data->deviceId, data->timestamp,
            data->current, data->voltage,
            data->power, data->energy,
            data->isComplete ? 1 : 0);

    fclose(fp);
    return true;
}

// 数据存储处理
void storeEnergyData() {
    printf("\n开始数据存储...\n");
    char filename[64];
    time_t now = time(NULL);
    struct tm* tmNow = localtime(&now);
    sprintf(filename, "energy_data_%04d%02d%02d.txt",
            tmNow->tm_year + 1900, tmNow->tm_mon + 1, tmNow->tm_mday);

    int successCount = 0;
    for (int i = 0; i < sysState.dataBufferIndex; i++) {
        if (saveDataToFile(&sysState.dataBuffer[i], filename)) {
            successCount++;
        } else {
            // 重试机制
            for (int j = 0; j < 2; j++) {
                printf("数据存储，重试第%d次...\n", j + 1);
                if (saveDataToFile(&sysState.dataBuffer[i], filename)) {
                    successCount++;
                    break;
                }
            }
        }
    }

    printf("数据存储完成: 共%d条，成功%d条，%d条\n",
           sysState.dataBufferIndex, successCount,
           sysState.dataBufferIndex - successCount);

    // 检查是否需要备份
    now = time(NULL);
    if (now - sysState.lastBackupTime >= BACKUP_INTERVAL) {
        printf("执行数据备份...\n");
        char backupFilename[64];
        sprintf(backupFilename, "backup_energy_%04d%02d%02d.txt",
                tmNow->tm_year + 1900, tmNow->tm_mon + 1, tmNow->tm_mday);
        
        // 备份操作
        FILE* src = fopen(filename, "r");
        FILE* dest = fopen(backupFilename, "w");
        if (src && dest) {
            char buf[1024];
            size_t len;
            while ((len = fread(buf, 1, sizeof(buf), src)) > 0) {
                fwrite(buf, 1, len, dest);
            }
            printf("数据备份完成: %s\n", backupFilename);
            sysState.lastBackupTime = now;
        } else {
            printf("数据备份\n");
        }
        if (src) fclose(src);
        if (dest) fclose(dest);
    }
}

// 从文件查询历史数据
void queryHistoryData(const char* deviceId, time_t start, time_t end) {
    printf("\n查询历史数据: 设备%s, 时间范围: %s至%s\n",
           deviceId, ctime(&start), ctime(&end));

    // 查询(实际应遍历备份文件)
    float totalEnergy = 0;
    int count = 0;
    for (int i = 0; i < sysState.dataBufferIndex; i++) {
        EnergyData* data = &sysState.dataBuffer[i];
        if (strcmp(data->deviceId, deviceId) == 0 &&
            data->timestamp >= start && data->timestamp <= end) {
            totalEnergy += data->energy;
            count++;
            if (count <= 5) {  // 只显示前5条
                printf("时间: %s  功率: %.2fW  用电量: %.2fWh\n",
                       ctime(&data->timestamp), data->power, data->energy);
            }
        }
    }

    if (count == 0) {
        printf("无符合条件的数据\n");
    } else {
        printf("共%d条记录，总用电量: %.2fWh\n", count, totalEnergy);
        if (count > 5) {
            printf("... 显示部分记录\n");
        }
    }
}

// 处理用户输入
void handleUserInput() {
    int choice;
    while (sysState.isRunning) {
        showMenu();
        scanf("%d", &choice);
        getchar();  // 吸收换行符

        switch (choice) {
            case 1:
                showDeviceList();
                break;
            case 2:
                showRealtimeData();
                break;
            case 3:
                showAlarmRecords();
                break;
            case 4: {
                int alarmIndex;
                showAlarmRecords();
                printf("请输入要处理的报警编号(0取消): ");
                scanf("%d", &alarmIndex);
                if (alarmIndex > 0 && alarmIndex <= sysState.alarmCount) {
                    handleAlarm(alarmIndex - 1);
                }
                break;
            }
            case 5: {
                char deviceId[32];
                int days;
                showDeviceList();
                printf("请输入设备ID: ");
                scanf("%s", deviceId);
                printf("请输入查询天数(1-30): ");
                scanf("%d", &days);
                if (days < 1 || days > 30) {
                    printf("无效的天数\n");
                    break;
                }
                time_t end = time(NULL);
                time_t start = end - days * 86400;
                queryHistoryData(deviceId, start, end);
                break;
            }
            case 6:
                generateEnergyReport();
                break;
            case 7: {
                int interval;
                printf("请输入采样间隔(%d-%d秒): ",
                       MIN_SAMPLING_INTERVAL, MAX_SAMPLING_INTERVAL);
                scanf("%d", &interval);
                setSamplingInterval(interval);
                break;
            }
            case 8: {
                float overload, highEnergy;
                printf("请输入过载阈值(W): ");
                scanf("%f", &overload);
                printf("请输入高能耗阈值(Wh): ");
                scanf("%f", &highEnergy);
                setAlarmThreshold(overload, highEnergy);
                break;
            }
            case 9:
                connectExternalSystem();
                break;
            case 0:
                sysState.isRunning = false;
                printf("系统已退出\n");
                break;
            default:
                printf("无效的操作\n");
        }
    }
}

// 系统主循环
void systemMainLoop() {
    while (sysState.isRunning) {
        // 数据采集
        collectEnergyData();
        
        // 数据传输
        transmitAllData();
        
        // 数据分析
        analyzeEnergyData();
        
        // 报警处理
        processAlarms();
        
        // 数据存储
        storeEnergyData();
        
        // 用户交互(每次循环后等待用户操作)
        handleUserInput();
        
        // 等待采样间隔时间
        if (sysState.isRunning) {
            printf("\n等待下次采集(%.2f分钟)...\n", 
                   (float)sysState.config.samplingInterval / 60);
            for (int i = 0; i < sysState.config.samplingInterval; i++) {
                if (!sysState.isRunning) break;
                sleep(1);
            }
        }
    }
}

// 添加用户
bool addUser(const char* username, const char* password, int level) {
    if (sysStateEx.userCount >= MAX_USER_NUM) {
        printf("用户数量已达上限\n");
        return false;
    }

    // 检查用户名是否已存在
    for (int i = 0; i < sysStateEx.userCount; i++) {
        if (strcmp(sysStateEx.users[i].username, username) == 0) {
            printf("用户名已存在\n");
            return false;
        }
    }

    UserInfo* user = &sysStateEx.users[sysStateEx.userCount];
    strcpy(user->username, username);
    // 简单加密存储密码(反转字符串)
    int pwdLen = strlen(password);
    for (int i = 0; i < pwdLen; i++) {
        user->password[i] = password[pwdLen - 1 - i];
    }
    user->password[pwdLen] = '\0';
    user->level = level;
    user->isActive = true;
    user->lastLogin = 0;

    sysStateEx.userCount++;
    printf("用户添加成功: %s\n", username);
    return true;
}

// 用户登录
bool userLogin(const char* username, const char* password) {
    for (int i = 0; i < sysStateEx.userCount; i++) {
        UserInfo* user = &sysStateEx.users[i];
        if (strcmp(user->username, username) == 0 && user->isActive) {
            // 验证密码(反转后比对)
            int pwdLen = strlen(password);
            char encryptedPwd[32] = {0};
            for (int j = 0; j < pwdLen; j++) {
                encryptedPwd[j] = password[pwdLen - 1 - j];
            }
            if (strcmp(user->password, encryptedPwd) == 0) {
                strcpy(sysStateEx.currentUser, username);
                user->lastLogin = time(NULL);
                printf("用户%s登录成功\n", username);
                return true;
            }
        }
    }
    printf("用户名或密码\n");
    return false;
}

// 检查用户权限
bool checkPermission(int requiredLevel) {
    if (strlen(sysStateEx.currentUser) == 0) {
        printf("请先登录系统\n");
        return false;
    }

    for (int i = 0; i < sysStateEx.userCount; i++) {
        if (strcmp(sysStateEx.users[i].username, sysStateEx.currentUser) == 0) {
            return sysStateEx.users[i].level <= requiredLevel;
        }
    }
    return false;
}

// 检查用电计划提醒
void checkPlanReminders() {
    time_t now = time(NULL);
    for (int i = 0; i < sysStateEx.planCount; i++) {
        PowerPlan* plan = &sysStateEx.plans[i];
        if (!plan->isEnabled) continue;

        // 计算计划是否在提醒时间范围内
        if (plan->startTime > now && plan->startTime - now <= PLAN_REMIND_TIME) {
            // 检查重复规则是否匹配
            struct tm* tmNow = localtime(&now);
            struct tm* tmPlan = localtime(&plan->startTime);
            bool match = false;

            switch (plan->repeat) {
                case 0:  // 不重复
                    match = (tmNow->tm_yday == tmPlan->tm_yday && 
                             tmNow->tm_year == tmPlan->tm_year);
                    break;
                case 1:  // 每天
                    match = true;
                    break;
                case 2:  // 工作日(周一至周五)
                    match = (tmNow->tm_wday >= 1 && tmNow->tm_wday <= 5);
                    break;
                case 3:  // 周末(周六至周日)
                    match = (tmNow->tm_wday == 0 || tmNow->tm_wday == 6);
                    break;
            }

            if (match) {
                int devIndex = findDeviceIndex(plan->deviceId);
                const char* devName = (devIndex != -1) ? sysStateEx.devices[devIndex].name : "未知设备";
                printf("\n[计划提醒] %s将在%.0f分钟后开始运行 - %s\n",
                       devName, (float)(plan->startTime - now) / 60, plan->remark);
            }
        }
    }
}

// 执行用电计划检查(开关设备)
void executePowerPlans() {
    time_t now = time(NULL);
    for (int i = 0; i < sysStateEx.planCount; i++) {
        PowerPlan* plan = &sysStateEx.plans[i];
        if (!plan->isEnabled) continue;

        int devIndex = findDeviceIndex(plan->deviceId);
        if (devIndex == -1) continue;

        struct tm* tmNow = localtime(&now);
        struct tm* tmPlanStart = localtime(&plan->startTime);
        struct tm* tmPlanEnd = localtime(&plan->endTime);

        // 检查重复规则是否匹配
        bool repeatMatch = false;
        switch (plan->repeat) {
            case 0:  // 不重复
                repeatMatch = (tmNow->tm_yday == tmPlanStart->tm_yday && 
                               tmNow->tm_year == tmPlanStart->tm_year);
                break;
            case 1:  // 每天
                repeatMatch = true;
                break;
            case 2:  // 工作日
                repeatMatch = (tmNow->tm_wday >= 1 && tmNow->tm_wday <= 5);
                break;
            case 3:  // 周末
                repeatMatch = (tmNow->tm_wday == 0 || tmNow->tm_wday == 6);
                break;
        }

        if (!repeatMatch) continue;

        // 检查是否在计划时间范围内
        bool inTimeRange = (now >= plan->startTime && now <= plan->endTime);
        DeviceInfo* dev = &sysStateEx.devices[devIndex];

        // 根据时间范围控制设备开关
        if (inTimeRange && !dev->status.isOn) {
            dev->status.isOn = true;
            printf("[计划执行] 开启设备: %s(%s) - %s\n",
                   dev->name, dev->id, plan->remark);
        } else if (!inTimeRange && dev->status.isOn) {
            dev->status.isOn = false;
            printf("[计划执行] 关闭设备: %s(%s) - %s\n",
                   dev->name, dev->id, plan->remark);
        }
    }
}

// 导出数据为CSV格式
bool exportDataCSV(ExportConfig* config, const char* filename) {
    FILE* fp = fopen(filename, "w");
    if (!fp) {
        printf("无法创建导出文件\n");
        return false;
    }

    // 写入CSV表头
    fprintf(fp, "设备ID,设备名称,时间,电流(A),电压(V),功率(W),用电量(Wh)\n");

    int count = 0;
    // 遍历缓冲区数据
    for (int i = 0; i < sysStateEx.dataBufferIndex && count < MAX_EXPORT_RECORDS; i++) {
        EnergyData* data = &sysStateEx.dataBuffer[i];
        // 检查设备过滤条件
        if (!config->includeAllDevices && 
            strcmp(data->deviceId, config->targetDeviceId) != 0) {
            continue;
        }
        // 检查时间范围
        if (data->timestamp < config->start || data->timestamp > config->end) {
            continue;
        }

        // 获取设备名称
        int devIndex = findDeviceIndex(data->deviceId);
        const char* devName = (devIndex != -1) ? sysStateEx.devices[devIndex].name : "未知";

        // 写入数据行
        fprintf(fp, "%s,%s,%s,%.2f,%.2f,%.2f,%.2f\n",
                data->deviceId, devName, ctime(&data->timestamp),
                data->current, data->voltage, data->power, data->energy);
        count++;
    }

    fclose(fp);
    printf("CSV数据导出完成: %s, 共%d条记录\n", filename, count);
    return true;
}

// 导出数据为JSON格式
bool exportDataJSON(ExportConfig* config, const char* filename) {
    FILE* fp = fopen(filename, "w");
    if (!fp) {
        printf("无法创建导出文件\n");
        return false;
    }

    fprintf(fp, "{\n");
    fprintf(fp, "  \"exportTime\": \"%s\",\n", ctime(&config->start));
    fprintf(fp, "  \"timeRange\": {\n");
    fprintf(fp, "    \"start\": \"%s\",\n", ctime(&config->start));
    fprintf(fp, "    \"end\": \"%s\"\n", ctime(&config->end));
    fprintf(fp, "  },\n");
    fprintf(fp, "  \"devices\": [\n");

    int devCount = 0;
    // 按设备分组导出
    for (int d = 0; d < sysStateEx.deviceCount; d++) {
        DeviceInfo* dev = &sysStateEx.devices[d];
        // 检查设备过滤条件
        if (!config->includeAllDevices && 
            strcmp(dev->id, config->targetDeviceId) != 0) {
            continue;
        }

        // 收集该设备的数据
        int dataCount = 0;
        if (devCount > 0) fprintf(fp, "    ,\n");
        fprintf(fp, "    {\n");
        fprintf(fp, "      \"deviceId\": \"%s\",\n", dev->id);
        fprintf(fp, "      \"deviceName\": \"%s\",\n", dev->name);
        fprintf(fp, "      \"data\": [\n");

        for (int i = 0; i < sysStateEx.dataBufferIndex && dataCount < MAX_EXPORT_RECORDS; i++) {
            EnergyData* data = &sysStateEx.dataBuffer[i];
            if (strcmp(data->deviceId, dev->id) != 0) continue;
            if (data->timestamp < config->start || data->timestamp > config->end) continue;

            if (dataCount > 0) fprintf(fp, "        ,\n");
            fprintf(fp, "        {\n");
            fprintf(fp, "          \"time\": \"%s\",\n", ctime(&data->timestamp));
            fprintf(fp, "          \"current\": %.2f,\n", data->current);
            fprintf(fp, "          \"voltage\": %.2f,\n", data->voltage);
            fprintf(fp, "          \"power\": %.2f,\n", data->power);
            fprintf(fp, "          \"energy\": %.2f\n", data->energy);
            fprintf(fp, "        }");
            dataCount++;
        }

        fprintf(fp, "\n      ]\n");
        fprintf(fp, "    }");
        devCount++;
    }

    fprintf(fp, "\n  ]\n");
    fprintf(fp, "}\n");
    fclose(fp);
    printf("JSON数据导出完成: %s, 涉及%d台设备\n", filename, devCount);
    return true;
}

// 执行数据导出
bool exportEnergyData(ExportConfig* config, const char* filename) {
    if (config->start >= config->end) {
        printf("导出时间范围无效\n");
        return false;
    }

    if (strcmp(config->format, "CSV") == 0) {
        return exportDataCSV(config, filename);
    } else if (strcmp(config->format, "JSON") == 0) {
        return exportDataJSON(config, filename);
    } else {
        printf("不支持的导出格式\n");
        return false;
    }
}

// 计算设备历史平均能耗
float calculateAverageEnergy(const char* deviceId, int days) {
    time_t end = time(NULL);
    time_t start = end - days * 86400;
    float total = 0;
    int count = 0;

    for (int i = 0; i < sysStateEx.dataBufferIndex; i++) {
        EnergyData* data = &sysStateEx.dataBuffer[i];
        if (strcmp(data->deviceId, deviceId) == 0 &&
            data->timestamp >= start && data->timestamp <= end) {
            total += data->energy;
            count++;
        }
    }

    return (count > 0) ? (total / days) : 0;
}

// 预测设备能耗
void predictDeviceEnergy(const char* deviceId) {
    // 基于过去7天的平均能耗进行预测
    float avgDaily = calculateAverageEnergy(deviceId, 7);
    if (avgDaily <= 0) {
        printf("没有足够数据进行能耗预测\n");
        return;
    }

    time_t now = time(NULL);
    // 清除该设备已有预测
    for (int i = 0; i < sysStateEx.predictionCount; ) {
        if (strcmp(sysStateEx.predictions[i].deviceId, deviceId) == 0) {
            // 移除该预测记录
            for (int j = i; j < sysStateEx.predictionCount - 1; j++) {
                sysStateEx.predictions[j] = sysStateEx.predictions[j + 1];
            }
            sysStateEx.predictionCount--;
        } else {
            i++;
        }
    }

    // 生成未来7天的预测
    for (int i = 1; i <= PREDICT_DAYS; i++) {
        if (sysStateEx.predictionCount >= PREDICT_DAYS * MAX_DEVICE_NUM) {
            printf("预测记录已达上限\n");
            break;
        }

        EnergyPrediction* pred = &sysStateEx.predictions[sysStateEx.predictionCount];
        strcpy(pred->deviceId, deviceId);
        pred->date = now + i * 86400;
        
        // 简单预测模型: 基础值±10%波动
        float fluctuation = (float)(rand() % 20 - 10) / 100; // -10% ~ +10%
        pred->predictedEnergy = avgDaily * (1 + fluctuation);
        pred->variance = avgDaily * 0.1; // 误差范围为平均值的10%

        sysStateEx.predictionCount++;
    }

    printf("设备%s能耗预测完成，预测未来%d天能耗\n", deviceId, PREDICT_DAYS);
}

// 显示能耗预测结果
void showEnergyPredictions(const char* deviceId) {
    printf("\n===== 能耗预测结果 =====\n");
    int count = 0;
    for (int i = 0; i < sysStateEx.predictionCount; i++) {
        EnergyPrediction* pred = &sysStateEx.predictions[i];
        if (deviceId != NULL && strcmp(pred->deviceId, deviceId) != 0) {
            continue;
        }

        int devIndex = findDeviceIndex(pred->deviceId);
        const char* devName = (devIndex != -1) ? sysStateEx.devices[devIndex].name : "未知设备";
        
        if (count == 0) {
            printf("设备: %s(%s)\n", devName, pred->deviceId);
            printf("日期       | 预测能耗(Wh) | 误差范围(±Wh)\n");
        } else if (deviceId == NULL && strcmp(pred->deviceId, sysStateEx.predictions[i-1].deviceId) != 0) {
            printf("\n设备: %s(%s)\n", devName, pred->deviceId);
            printf("日期       | 预测能耗(Wh) | 误差范围(±Wh)\n");
        }

        struct tm* tmDate = localtime(&pred->date);
        char dateStr[16];
        sprintf(dateStr, "%04d-%02d-%02d",
                tmDate->tm_year + 1900, tmDate->tm_mon + 1, tmDate->tm_mday);
        printf("%s | %.2f        | %.2f\n",
               dateStr, pred->predictedEnergy, pred->variance);
        count++;
    }

    if (count == 0) {
        printf("无能耗预测数据，请先执行预测\n");
    }
}

// 处理用电计划操作
void handlePowerPlanOperations() {
    int choice;
    while (true) {
        showPlanSubMenu();
        scanf("%d", &choice);
        getchar();

        switch (choice) {
            case 1:
                showPowerPlans();
                break;
            case 2: {
                if (!checkPermission(USER_LEVEL)) {
                    printf("权限不足，无法添加用电计划\n");
                    break;
                }
                
                char deviceId[32], remark[128];
                int startHour, startMin, endHour, endMin, repeat;
                showDeviceList();
                printf("请输入设备ID: ");
                scanf("%s", deviceId);
                printf("请输入开始时间(小时 分钟): ");
                scanf("%d %d", &startHour, &startMin);
                printf("请输入结束时间(小时 分钟): ");
                scanf("%d %d", &endHour, &endMin);
                printf("请选择重复方式(0-不重复,1-每天,2-工作日,3-周末): ");
                scanf("%d", &repeat);
                getchar(); // 吸收换行
                printf("请输入计划备注: ");
                fgets(remark, sizeof(remark), stdin);
                // 去除备注中的换行符
                remark[strcspn(remark, "\n")] = '\0';

                // 构建时间戳(使用今天的日期)
                time_t now = time(NULL);
                struct tm* tmNow = localtime(&now);
                struct tm tmStart = *tmNow;
                struct tm tmEnd = *tmNow;

                tmStart.tm_hour = startHour;
                tmStart.tm_min = startMin;
                tmStart.tm_sec = 0;
                tmEnd.tm_hour = endHour;
                tmEnd.tm_min = endMin;
                tmEnd.tm_sec = 0;

                // 如果结束时间早于开始时间，视为第二天
                if (mktime(&tmEnd) < mktime(&tmStart)) {
                    tmEnd.tm_mday += 1;
                }

                addPowerPlan(deviceId, mktime(&tmStart), mktime(&tmEnd), repeat, remark);
                break;
            }
            case 3: {
                int planIndex;
                showPowerPlans();
                printf("请输入要修改状态的计划编号(0取消): ");
                scanf("%d", &planIndex);
                if (planIndex > 0 && planIndex <= sysStateEx.planCount) {
                    PowerPlan* plan = &sysStateEx.plans[planIndex - 1];
                    plan->isEnabled = !plan->isEnabled;
                    printf("计划%s已%s\n", plan->id, plan->isEnabled ? "启用" : "禁用");
                }
                break;
            }
            case 4: {
                if (!checkPermission(ADMIN_LEVEL)) {
                    printf("权限不足，无法删除用电计划\n");
                    break;
                }
                
                int planIndex;
                showPowerPlans();
                printf("请输入要删除的计划编号(0取消): ");
                scanf("%d", &planIndex);
                if (planIndex > 0 && planIndex <= sysStateEx.planCount) {
                    // 移除计划
                    for (int i = planIndex - 1; i < sysStateEx.planCount - 1; i++) {
                        sysStateEx.plans[i] = sysStateEx.plans[i + 1];
                    }
                    sysStateEx.planCount--;
                    printf("计划已删除\n");
                }
                break;
            }
            case 0:
                return;
            default:
                printf("无效的操作\n");
        }
    }
}

        scanf("%s", format);
        printf("请输入时间范围(天数，0表示今天):\n");
        printf("开始天数: ");
        scanf("%d", &startDays);
        printf("结束天数: ");
        scanf("%d", &endDays);

        // 计算时间范围
        time_t now = time(NULL);
        config.start = now - startDays * 86400;
        config.end = now - endDays * 86400;
        if (config.start > config.end) {
            printf("开始时间不能晚于结束时间\n");
            continue;
        }

        // 设置格式
        strcpy(config.format, format);
        config.includeAllDevices = (choice == 1);

        if (choice == 2) {
            showDeviceList();
            printf("请输入设备ID: ");
            scanf("%s", config.targetDeviceId);
        }

        printf("请输入导出文件名: ");
        scanf("%s", filename);

        // 执行导出
        exportEnergyData(&config, filename);
    }
}

// 系统主循环
void extendedSystemMainLoop() {
    while (sysStateEx.isRunning) {
        // 数据采集
        collectEnergyData();
        
        // 数据传输
        transmitAllData();
        
        // 数据分析
        analyzeEnergyData();
        
        // 报警处理
        processAlarms();
        
        // 设备维护提醒
        checkMaintenanceReminders();
        
        // 用电计划提醒
        checkPlanReminders();
        
        // 执行用电计划
        executePowerPlans();
        
        // 数据存储
        storeEnergyData();
        
        // 用户交互
        handleExtendedUserInput();
        
        // 等待采样间隔时间
        if (sysStateEx.isRunning) {
            printf("\n等待下次采集(%.2f分钟)...\n", 
                   (float)sysStateEx.config.samplingInterval / 60);
            for (int i = 0; i < sysStateEx.config.samplingInterval; i++) {
                if (!sysStateEx.isRunning) break;
                sleep(1);
            }
        }
    }
}

void initSystemExt() {
    memset(&sysState, 0, sizeof(SystemStateExt));
    sysState.deviceCount = 0;
    sysState.dataBufferIndex = 0;
    sysState.alarmCount = 0;
    sysState.planCount = 0;
    sysState.userCount = 0;
    sysState.predictionCount = 0;
    sysState.groupCount = 0;
    sysState.cmdCount = 0;
    sysState.logCount = 0;
    sysState.lastBackupTime = time(NULL);
    sysState.isRunning = true;
    memset(sysState.currentUser, 0, sizeof(sysState.currentUser));

    // 初始化默认配置
    sysState.config.samplingInterval = DEFAULT_SAMPLING_INTERVAL;
    sysState.config.overloadThreshold = 2000;
    sysState.config.highEnergyThreshold = ENERGY_THRESHOLD_HIGH;
    sysState.config.notifyByApp = true;
    sysState.config.notifyBySms = true;
    strcpy(sysState.config.mqttServer, "mqtt://energy-server.local:1883");

    // 初始化MQTT配置
    sprintf(sysState.mqttConfig.clientId, "energy-client-%ld", time(NULL));
    strcpy(sysState.mqttConfig.username, "energy-user");
    strcpy(sysState.mqttConfig.password, "mqtt-pass");
    sysState.mqttConfig.port = 1883;
    sysState.mqttConfig.cleanSession = true;
    sysState.mqttConfig.lastHeartbeat = 0;

    // 初始化预测模型权重(前7天权重递减)
    for (int i = 0; i < MAX_DEVICE_NUM; i++) {
        float sum = 0;
        for (int j = 0; j < PREDICT_WINDOW; j++) {
            sysState.predModels[i].weights[j] = PREDICT_WINDOW - j;
            sum += sysState.predModels[i].weights[j];
        }
        // 权重归一化
        for (int j = 0; j < PREDICT_WINDOW; j++) {
            sysState.predModels[i].weights[j] /= sum;
        }
        sysState.predModels[i].bias = 0.1;
        sysState.predModels[i].errorRate = 0.15;
    }

    // 添加默认管理员
    addUser("admin", "admin123", ADMIN_LEVEL);
    printf("系统初始化完成\n");
}

// 创建设备分组
bool createDeviceGroup(const char* groupName) {
    if (sysState.groupCount >= MAX_GROUP_NUM) {
        printf("分组数量已达上限\n");
        return false;
    }

    DeviceGroup* group = &sysState.groups[sysState.groupCount];
    sprintf(group->groupId, "GRP%02d", sysState.groupCount + 1);
    strcpy(group->groupName, groupName);
    group->deviceCount = 0;
    memset(group->deviceIds, 0, sizeof(group->deviceIds));

    sysState.groupCount++;
    printf("设备分组创建成功: %s(%s)\n", groupName, group->groupId);
    return true;
}

// 向分组添加设备
bool addDeviceToGroup(const char* groupId, const char* deviceId) {
    int groupIndex = -1;
    for (int i = 0; i < sysState.groupCount; i++) {
        if (strcmp(sysState.groups[i].groupId, groupId) == 0) {
            groupIndex = i;
            break;
        }
    }
    if (groupIndex == -1) {
        printf("分组不存在\n");
        return false;
    }

    DeviceGroup* group = &sysState.groups[groupIndex];
    if (group->deviceCount >= MAX_DEVICE_NUM) {
        printf("分组设备数量已达上限\n");
        return false;
    }

    // 检查设备是否存在
    if (findDeviceIndex(deviceId) == -1) {
        printf("设备不存在\n");
        return false;
    }

    // 检查是否已在分组中
    for (int i = 0; i < group->deviceCount; i++) {
        if (strcmp(group->deviceIds[i], deviceId) == 0) {
            printf("设备已在该分组中\n");
            return false;
        }
    }

    strcpy(group->deviceIds[group->deviceCount], deviceId);
    group->deviceCount++;
    printf("设备%s已添加到分组%s\n", deviceId, group->groupName);
    return true;
}

// 显示设备分组及能耗
void showGroupEnergy() {
    printf("\n===== 分组能耗统计 =====\n");
    for (int g = 0; g < sysState.groupCount; g++) {
        DeviceGroup* group = &sysState.groups[g];
        float totalEnergy = 0;
        printf("分组: %s(%s)\n", group->groupName, group->groupId);
        printf("包含设备: ");
        for (int d = 0; d < group->deviceCount; d++) {
            char* deviceId = group->deviceIds[d];
            int devIndex = findDeviceIndex(deviceId);
            if (devIndex == -1) continue;
            
            DeviceInfo* dev = &sysState.devices[devIndex];
            printf("%s(%s) ", dev->name, deviceId);
            totalEnergy += dev->status.energy;
        }
        printf("\n总能耗: %.2fWh\n", totalEnergy);    }
}

// 记录设备操作日志
void logDeviceAction(const char* deviceId, const char* operator, const char* action, bool success) {
    if (sysState.logCount >= MAX_DEVICE_LOG) {
        // 日志满时移除最早记录
        memmove(sysState.deviceLogs, sysState.deviceLogs + 1, 
               (MAX_DEVICE_LOG - 1) * sizeof(DeviceLog));
        sysState.logCount--;
    }

    DeviceLog* log = &sysState.deviceLogs[sysState.logCount];
    strcpy(log->deviceId, deviceId);
    log->opTime = time(NULL);
    strcpy(log->operator, operator);
    strcpy(log->action, action);
    log->success = success;

    sysState.logCount++;
}

// 发送远程控制指令
bool sendRemoteCmd(const char* deviceId, bool targetState) {
    if (sysState.cmdCount >= 32) {
        printf("指令队列已满\n");
        return false;
    }

    int devIndex = findDeviceIndex(deviceId);
    if (devIndex == -1) {
        printf("设备不存在\n");
        return false;
    }

    RemoteCmd* cmd = &sysState.cmdQueue[sysState.cmdCount];
    strcpy(cmd->deviceId, deviceId);
    cmd->targetState = targetState;
    cmd->sendTime = time(NULL);
    cmd->isExecuted = false;
    strcpy(cmd->result, "未执行");

    sysState.cmdCount++;
    printf("远程指令已发送: 设备%s, 目标状态%s\n", 
           deviceId, targetState ? "开启" : "关闭");
    return true;
}

// 处理远程控制指令
void processRemoteCmds() {
    time_t now = time(NULL);
    for (int i = 0; i < sysState.cmdCount; i++) {
        RemoteCmd* cmd = &sysState.cmdQueue[i];
        if (cmd->isExecuted) continue;

        // 检查指令是否超时
        if (now - cmd->sendTime > REMOTE_CMD_TIMEOUT) {
            strcpy(cmd->result, "指令超时");
            cmd->isExecuted = true;
            logDeviceAction(cmd->deviceId, sysState.currentUser, 
                           cmd->targetState ? "远程开启" : "远程关闭", false);
            continue;
        }

        // 执行指令
        int devIndex = findDeviceIndex(cmd->deviceId);
        if (devIndex == -1) {
            strcpy(cmd->result, "设备不存在");
            cmd->isExecuted = true;
            continue;
        }

        DeviceInfo* dev = &sysState.devices[devIndex];
        dev->status.isOn = cmd->targetState;
        dev->status.lastUpdate = now;
        cmd->isExecuted = true;
        strcpy(cmd->result, "执行成功");
        
        logDeviceAction(cmd->deviceId, sysState.currentUser, 
                       cmd->targetState ? "远程开启" : "远程关闭", true);
        printf("远程指令执行成功: 设备%s已%s\n", 
               cmd->deviceId, cmd->targetState ? "开启" : "关闭");
    }

    // 清理已执行指令
    int newCount = 0;
    for (int i = 0; i < sysState.cmdCount; i++) {
        if (!sysState.cmdQueue[i].isExecuted) {
            sysState.cmdQueue[newCount] = sysState.cmdQueue[i];
            newCount++;
        }
    }
    sysState.cmdCount = newCount;
}

// 优化的能耗预测模型
void optimizePredictionModel(const char* deviceId) {
    int devIndex = findDeviceIndex(deviceId);
    if (devIndex == -1) {
        printf("设备不存在\n");
        return;
    }

    // 获取过去14天的能耗数据用于模型训练
    time_t end = time(NULL);
    time_t start = end - 14 * 86400;
    float dailyEnergy[14] = {0};
    int dayCount = 0;

    // 按天统计能耗
    for (int i = 0; i < sysState.dataBufferIndex; i++) {
        EnergyData* data = &sysState.dataBuffer[i];
        if (strcmp(data->deviceId, deviceId) != 0) continue;
        if (data->timestamp < start || data->timestamp > end) continue;

        struct tm* tmData = localtime(&data->timestamp);
        int dayOffset = (end - data->timestamp) / 86400;
        if (dayOffset >= 0 && dayOffset < 14) {
            dailyEnergy[dayOffset] += data->energy;
            dayCount++;
        }
    }

    if (dayCount == 0) {
        printf("无足够数据优化模型\n");
        return;
    }

    // 计算模型权重(最近7天权重更高)
    PredictionModel* model = &sysState.predModels[devIndex];
    strcpy(model->deviceId, deviceId);
    float sum = 0;
    for (int i = 0; i < PREDICT_WINDOW; i++) {
        model->weights[i] = (i + 1) * (i + 1);  // 权重平方增长
        sum += model->weights[i];
    }
    // 权重归一化
    for (int i = 0; i < PREDICT_WINDOW; i++) {
        model->weights[i] /= sum;
    }

    // 计算误差率
    float actualAvg = 0;
    for (int i = 0; i < 7; i++) actualAvg += dailyEnergy[i];
    actualAvg /= 7;

    float predAvg = 0;
    for (int i = 0; i < 7; i++) predAvg += dailyEnergy[i + 7] * model->weights[i];
    
    model->errorRate = fabs(predAvg - actualAvg) / actualAvg;
    model->bias = 0.05 * actualAvg;  // 偏置为平均能耗的5%
    printf("设备%s预测模型优化完成，误差率: %.2f%%\n", 
           deviceId, model->errorRate * 100);
}

// 使用优化模型预测能耗
void predictWithModel(const char* deviceId) {
    int devIndex = findDeviceIndex(deviceId);
    if (devIndex == -1) {
        printf("设备不存在\n");
        return;
    }

    // 先优化模型
    optimizePredictionModel(deviceId);
    PredictionModel* model = &sysState.predModels[devIndex];

    // 获取过去7天的能耗数据
    time_t end = time(NULL);
    time_t start = end - 7 * 86400;
    float recentEnergy[7] = {0};

    for (int i = 0; i < sysState.dataBufferIndex; i++) {
        EnergyData* data = &sysState.dataBuffer[i];
        if (strcmp(data->deviceId, deviceId) != 0) continue;
        if (data->timestamp < start || data->timestamp > end) continue;

        struct tm* tmData = localtime(&data->timestamp);
        int dayOffset = (end - data->timestamp) / 86400;
        if (dayOffset >= 0 && dayOffset < 7) {
            recentEnergy[dayOffset] += data->energy;
        }
    }

    // 清除该设备已有预测
    for (int i = 0; i < sysState.predictionCount; ) {
        if (strcmp(sysState.predictions[i].deviceId, deviceId) == 0) {
            for (int j = i; j < sysState.predictionCount - 1; j++) {
                sysState.predictions[j] = sysState.predictions[j + 1];
            }
            sysState.predictionCount--;
        } else {
            i++;
        }
    }

    // 生成未来7天预测
    time_t now = time(NULL);
    for (int d = 1; d <= PREDICT_DAYS; d++) {
        if (sysState.predictionCount >= PREDICT_DAYS * MAX_DEVICE_NUM) break;

        EnergyPrediction* pred = &sysState.predictions[sysState.predictionCount];
        strcpy(pred->deviceId, deviceId);
        pred->date = now + d * 86400;

        // 加权预测计算
        pred->predictedEnergy = model->bias;
        for (int i = 0; i < PREDICT_WINDOW; i++) {
            pred->predictedEnergy += recentEnergy[i] * model->weights[i];
        }

        // 加入随机波动(基于误差率)
        float fluctuation = (float)(rand() % (int)(model->errorRate * 200) - 
                                   (int)(model->errorRate * 100)) / 100;
        pred->predictedEnergy *= (1 + fluctuation);
        pred->variance = pred->predictedEnergy * model->errorRate;

        sysState.predictionCount++;
    }

    printf("设备%s能耗预测完成(基于优化模型)\n", deviceId);
}

// 增量数据备份
bool incrementalBackup() {
    time_t now = time(NULL);
    char backupDir[64] = "energy_backup/";
    mkdir(backupDir, 0755);  // 创建备份目录

    // 计算当前数据与上次备份的差异率
    float changeRate = 0;
    int totalRecords = sysState.dataBufferIndex;
    if (totalRecords > 0) {
        // 简单: 随机生成变化率(实际应比较数据哈希)
        changeRate = (float)(rand() % 50) / 100;  // 0-50%
    }

    // 达到阈值则执行增量备份
    if (changeRate < INCREMENTAL_BACKUP_RATE && sysState.lastBackupTime != 0) {
        printf("数据变化率%.2f%%，未达到增量备份阈值(%.0f%%)\n", 
               changeRate * 100, INCREMENTAL_BACKUP_RATE * 100);
        return false;
    }

    // 生成备份文件名
    struct tm* tmNow = localtime(&now);
    char filename[128];
    sprintf(filename, "%senergy_inc_%04d%02d%02d_%02d%02d%02d.txt",
            backupDir, tmNow->tm_year + 1900, tmNow->tm_mon + 1, tmNow->tm_mday,
            tmNow->tm_hour, tmNow->tm_min, tmNow->tm_sec);

    FILE* fp = fopen(filename, "w");
    if (!fp) {
        printf("增量备份: 无法创建文件\n");
        return false;
    }

    // 只备份变化的数据(这里简化为备份缓冲区所有数据)
    int backupCount = 0;
    for (int i = 0; i < sysState.dataBufferIndex; i++) {
        EnergyData* data = &sysState.dataBuffer[i];
        if (data->timestamp > sysState.lastBackupTime) {  // 只备份新数据
            fprintf(fp, "%s,%ld,%.2f,%.2f,%.2f,%.2f,%d\n",
                    data->deviceId, data->timestamp,
                    data->current, data->voltage,
                    data->power, data->energy,
                    data->isComplete ? 1 : 0);
            backupCount++;
        }
    }

    fclose(fp);
    sysState.lastBackupTime = now;
    printf("增量备份完成: %s，备份记录数: %d\n", filename, backupCount);
    return true;
}

// 多协议通讯适配(根据距离选择)
CommType adaptCommType(const char* deviceId, float distance) {
    // 距离单位: 米
    if (distance < 10) {
        // 近距离优先蓝牙
        if (isCommAvailable(COMM_BLUETOOTH)) return COMM_BLUETOOTH;
    } else if (distance < 50) {
        // 中距离优先Zigbee
        if (isCommAvailable(COMM_ZIGBEE)) return COMM_ZIGBEE;
    }
    // 远距离或其他情况用WiFi
    return COMM_WIFI;
}

// 按分组采集数据
void collectGroupData(const char* groupId) {
    int groupIndex = -1;
    for (int i = 0; i < sysState.groupCount; i++) {
        if (strcmp(sysState.groups[i].groupId, groupId) == 0) {
            groupIndex = i;
            break;
        }
    }
    if (groupIndex == -1) {
        printf("分组不存在\n");
        return;
    }

    DeviceGroup* group = &sysState.groups[groupIndex];
    printf("\n开始采集分组%s数据...\n", group->groupName);
    for (int d = 0; d < group->deviceCount; d++) {
        char* deviceId = group->deviceIds[d];
        int devIndex = findDeviceIndex(deviceId);
        if (devIndex == -1) continue;

        DeviceInfo* dev = &sysState.devices[devIndex];
        printf("处理设备: %s ", dev->name);

        if (!sensorSelfCheck(deviceId)) {
            dev->status.isFault = true;
            printf("[传感器故障]\n");
            continue;
        }

        DeviceStatus newStatus = collectDeviceData(deviceId);
        // 数据滤波
        newStatus.current = filterData(newStatus.current, dev->status.current);
        newStatus.voltage = filterData(newStatus.voltage, dev->status.voltage);
        newStatus.power = filterData(newStatus.power, dev->status.power);

        dev->status = newStatus;

        // 存入缓冲区
        EnergyData data;
        strcpy(data.deviceId, deviceId);
        data.timestamp = time(NULL);
        data.current = newStatus.current;
        data.voltage = newStatus.voltage;
        data.power = newStatus.power;
        data.energy = newStatus.energy;
        data.isComplete = true;
        convertDataFormat(&data);

        if (sysState.dataBufferIndex < MAX_DATA_BUFFER) {
            sysState.dataBuffer[sysState.dataBufferIndex] = data;
            sysState.dataBufferIndex++;
        } else {
            memmove(sysState.dataBuffer, sysState.dataBuffer + 1, 
                   (MAX_DATA_BUFFER - 1) * sizeof(EnergyData));
            sysState.dataBuffer[MAX_DATA_BUFFER - 1] = data;
        }

        printf("[采集完成, 功率: %.2fW]\n", data.power);
    }
    printf("分组数据采集结束\n");
}

// 报警类型: 用电计划偏离报警
void checkPlanDeviation() {
    time_t now = time(NULL);
    // 检查当前运行设备是否符合计划
    for (int i = 0; i < sysState.deviceCount; i++) {
        DeviceInfo* dev = &sysState.devices[i];
        if (!dev->status.isOn) continue;

        // 检查该设备是否有计划限制
        bool shouldBeOn = false;
        for (int p = 0; p < sysState.planCount; p++) {
            PowerPlan* plan = &sysState.plans[p];
            if (strcmp(plan->deviceId, dev->id) != 0 || !plan->isEnabled) continue;

            // 检查当前是否在计划时间内
            struct tm* tmNow = localtime(&now);
            struct tm* tmPlanStart = localtime(&plan->startTime);
            bool inPlanTime = false;

            switch (plan->repeat) {
                case 1:  // 每天
                    inPlanTime = (tmNow->tm_hour > tmPlanStart->tm_hour ||
                                 (tmNow->tm_hour == tmPlanStart->tm_hour && 
                                  tmNow->tm_min >= tmPlanStart->tm_min)) &&
                                 (tmNow->tm_hour < localtime(&plan->endTime)->tm_hour ||
                                  (tmNow->tm_hour == localtime(&plan->endTime)->tm_hour && 
                                   tmNow->tm_min <= localtime(&plan->endTime)->tm_min));
                    shouldBeOn = inPlanTime;
                    break; 
            }
        }
 
        if (!shouldBeOn) {
            char desc[128];
            sprintf(desc, "%s在非计划时间运行，可能导致额外能耗", dev->name);
            addAlarm(ALARM_HIGH_ENERGY, dev->id, desc);
        }
    }
}
